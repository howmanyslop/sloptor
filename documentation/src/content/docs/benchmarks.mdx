export const cold = getReport("cold").comparisons;
export const smallest = cold[0];
export const largest = cold[cold.length - 1];
export const rbxtscFloor = formatMilliseconds(getRival(smallest, "rbxtsc").median);
export const sloptorLargest = formatMilliseconds(largest.sloptor.median);

export const tableNote = `fig. 2 — the median of each run, in milliseconds, for when you want to read the numbers off. A cold build of ${smallest.project}, the one-file project, takes rbxtsc ${rbxtscFloor} ms. sloptor builds ${largest.project}, the largest, in ${sloptorLargest} ms.`;
